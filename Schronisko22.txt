    function vetAction(id, type) {
        const v = state.vets.find(vet => vet.id === id);
        if(type === 'urlop') {
            v.busyUntil = Date.now() + (48 * 60 * 60 * 1000);
            v.busyType = "urlop";
            v.mood = Math.min(v.mood + 40, 100);
            alert(v.name + " odpoczywa przez 48h.");
        } else if(type === 'szkolenie') {
            const prices = { 1: 500, 2: 1000, 3: 1500, 4: 2000 };
            const cost = prices[v.exp] || 2500;
            if(state.money < cost) return alert("Brak środków!");

            v.busyType = "szkolenie";

    function prepJob(idx, type) {
        const dog = state.boxes[state.currentBox].dogs[idx];
        let dur = 0;
        let workerName = "";

        if(type === 'Klinika') {
            if(state.inventory.medical <= 0) return alert("Brak apteczek!");
            const vet = state.vets.find(v => v.busyUntil < Date.now());
            if(!vet) return alert("Brak wolnych weterynarzy!");
            
            const caregiver = state.boxes[state.currentBox].assigned.find(w => w !== "" && (!state.busyUntil[w] || state.busyUntil[w] < Date.now()));
            if(!caregiver) return alert("Opiekun boksowy musi być wolny, by przekazać psa!");

            state.inventory.medical--;
            dur = (100 - dog.health) * 120000;
            vet.busyUntil = Date.now() + dur; 
            vet.busyType = "praca";
            state.busyUntil[caregiver] = Date.now() + 5000; 
            workerName = vet.name;
        } else {
            const caregiver = state.boxes[state.currentBox].assigned.find(w => w !== "" && (!state.busyUntil[w] || state.busyUntil[w] < Date.now()));
            if(!caregiver) return alert("Brak wolnych opiekunów!");
            
            if(type==='Karm') {
                if(state.inventory.food < 1.0) return alert("Brak karmy!");
                state.inventory.food -= 1.0;
                dur = (100-dog.hunger)*(900000/50);
            }
            else if(type==='Sprzątaj') {
                if(state.inventory.cleaner < 0.2) return alert("Brak płynu!");
                state.inventory.cleaner -= 0.2;
                dur = (100-dog.clean)*60000;
            }
            
            state.busyUntil[caregiver] = Date.now() + dur;
            workerName = caregiver;
        }
        
        dog.job = { type, endTime: Date.now() + dur, duration: dur, worker: workerName };
        
// --- TO DODAJE PAMIĘĆ STAŁĄ ---
localStorage.setItem('schronisko_save', JSON.stringify(state));

renderDogs();
updateStockDisplay();
    }

function calculateOfflineCosts() {
    const now = new Date(); // Aktualny czas
    const lastLogin = localStorage.getItem('lastLoginDate'); // Pobieramy datę ostatniej wizyty
    
    if (lastLogin) {
        const lastDate = new Date(lastLogin);
        // Obliczamy różnicę w dniach
        const diffInTime = now.getTime() - lastDate.getTime();
        const diffInDays = Math.floor(diffInTime / (1000 * 3600 * 24));

        if (diffInDays > 0) {
            const dailyWage = 50; // Stawka za 1 pracownika
            const staffCount = state.staff.length;
            const totalToSubtract = diffInDays * staffCount * dailyWage;

            state.stats.money -= totalToSubtract;
            state.stats.daysPassed += diffInDays;
            
            alert("Podczas Twojej nieobecności minęło dni: " + diffInDays + ". Koszty kadry: -" + totalToSubtract + " zł.");
        }
    }
    
    // Zapisujemy obecny czas jako ostatni login
    localStorage.setItem('lastLoginDate', now.toISOString());
}

function processAdoption(dogId, home, family, pets) {
    const dog = state.dogs.find(d => d.id === dogId);
    let isBadMatch = false;

    // Przykładowe konflikty:
    if (dog.traits.env === "Potrzebuje ogrodu" && home === "Małe mieszkanie w bloku") isBadMatch = true;
    if (dog.traits.soc === "Tylko dla dorosłych" && family === "Ma małe dzieci") isBadMatch = true;
    if (dog.traits.soc === "Goni koty" && pets === "Ma koty") isBadMatch = true;
    if (dog.traits.spec === "Lęk separacyjny" && family === "Mieszka sam") isBadMatch = true;

    // Realizacja adopcji
    state.stats.money += 200;
    
    if (isBadMatch) {
        // Pies wróci po 5 dniach (dla testów możesz zmienić na 5 minut!)
        const returnDate = new Date();
        returnDate.setDate(returnDate.getDate() + 5); 
        
        // Tutaj można by dodać psa do listy "oczekujących na powrót" w localStorage
        alert("Adopcja udana! Otrzymano 200 zł. (Uwaga: Coś mi mówi, że to nie był idealny wybór...)");
    } else {
        alert("Wspaniała decyzja! Pies znalazł idealny dom. +200 zł");
    }

    // Usuwamy psa ze schroniska
    state.dogs = state.dogs.filter(d => d.id !== dogId);
    
    // Zamykamy okno i odświeżamy widoki
    document.getElementById('candidates-overlay').style.display = 'none';
    refreshAdoptionView();
    if(typeof updateUI === 'function') updateUI(); // Odśwież pieniądze na ekranie
}

function nextStep() {

            desc.innerText = introMessages[currentStep].d;

        document.getElementById('intro-text-step').style.display = 'none';

function startSearch(type) {
    const config = searchConfig[type];
    if (localStorage.getItem('search_end_' + type)) {
        alert("Patrol już tam jest!");
        return;
    }

    alert(config.msg);
    const endTime = Date.now() + config.time;
    localStorage.setItem('search_end_' + type, endTime);
    updateOfficeTimers();
}

function updateOfficeTimers() {
    const types = ['forest', 'city', 'home'];
    const names = { forest: '🌳 Las', city: '🏙️ Miasto', home: '🏠 Dom' };
    const timeNames = { forest: '4h', city: '2h', home: '1h' };
    
    types.forEach(type => {
        const endTime = localStorage.getItem('search_end_' + type);
        const display = document.getElementById('timer-' + type);
        
        if (!display) return;

        if (endTime) {
            const remaining = endTime - Date.now();
            if (remaining <= 0) {
                // PRZYCISK ODBIERANIA
                display.innerHTML = `<button onclick="collectDogs('${type}')" style="background:#ff9800; color:white; border:none; border-radius:5px; padding:10px; width:100%; cursor:pointer; font-weight:bold;">ODBIERZ ZWIERZAKI 🔍</button>`;
            } else {
                // LICZNIK CZASU
                const h = Math.floor(remaining / 3600000);
                const m = Math.floor((remaining % 3600000) / 60000);
                const s = Math.floor((remaining % 60000) / 1000);
                display.innerText = `W trasie: ${h}h ${m}m ${s}s`;
            }
        } else {
            // PRZYCISK STARTU (Teraz klikamy w przycisk, a nie w diva!)
            display.innerHTML = `<button onclick="startSearch('${type}')" style="background:rgba(255,255,255,0.2); color:white; border:1px solid rgba(255,255,255,0.3); border-radius:5px; padding:10px; width:100%; text-align:left; cursor:pointer;">${names[type]} (${timeNames[type]}) - SZUKAJ</button>`;
        }
    });
}

function renameDog(dogId) {
    const newName = prompt("Jak chcesz nazwać tego uratowanego psiaka?");
    if (!newName || newName.trim() === "") return;

    let found = false;
    state.boxes.forEach(box => {
        // Szukamy psa po ID lub po indexie
        const dog = box.dogs.find(d => d.id == dogId) || box.dogs[dogId];
        if (dog) {
            dog.name = newName.trim();
            found = true;
        }
    });

    if (found) {
        saveGame(); // ZAPISUJEMY!
        renderDogs(); // ODŚWIEŻAMY WIDOK!
        alert("Imię zostało zmienione!");
    } else {
        alert("Nie znaleziono pieska do zmiany imienia.");
    }
}

    // 1. CZYŚCIMY PAMIĘĆ NATYCHMIAST
    localStorage.removeItem('search_end_' + type);
    
    // 2. WYMUSZAMY ZMIANĘ PRZYCISKU NA "SZUKAJ" ZANIM WYSKOCZY ALERT
    // Dzięki temu, nawet jak będziesz długo trzymać okienko alert, 
    // system pod spodem już wie, że misja skończona.
    if (typeof updateOfficeTimers === 'function') updateOfficeTimers();

    // 3. LOSOWANIE
    let found = (type === 'home') ? (Math.random() > 0.5 ? 2 : 1) : (Math.floor(Math.random() * 3) + 1);
    
    // 4. KOMUNIKAT
    alert("Patrol zakończony! Znaleziono: " + found);

    // 5. PRÓBA DODANIA PSÓW (Bezpieczna)
    try {
        if (typeof addNewDog === 'function') {
            for (let i = 0; i < found; i++) {
                addNewDog();
            }
        }
    } catch (e) {
        console.error("Błąd dodawania psa, ale pętla powinna być już przerwana:", e);
    }

    // 6. OSTATECZNE ODŚWIEŻENIE WSZYSTKIEGO
    if (typeof updateOfficeTimers === 'function') updateOfficeTimers();
    if (typeof renderDogs === 'function') renderDogs();
    if (typeof saveGame === 'function') saveGame();
}

function processJobs() {
    let stateChanged = false;
    
    state.boxes.forEach(box => {
        box.dogs.forEach(dog => {
            // Jeśli pies ma przypisaną pracę i czas tej pracy właśnie minął...
            if (dog.job && Date.now() >= dog.job.endTime) {
                
                // Dajemy psu nagrodę za cierpliwość (30% -> 100%)
                if (dog.job.type === 'Karm') dog.hunger = 100;
                if (dog.job.type === 'Sprzątaj') dog.clean = 100;
                if (dog.job.type === 'Lecz') dog.health = 100;

                dog.job = null; // Pies kończy pracę
                stateChanged = true;
                console.log("Praca zakończona pomyślnie!");
            }
        });
    });

    if (stateChanged) {
        localStorage.setItem('schronisko_save', JSON.stringify(state));
        renderDogs(); // Odświeżamy widok boksów
    }
}

// Uruchamiamy sprawdzanie co 1 sekundę
setInterval(processJobs, 1000);

const originalSave = saveGame; // robimy kopię Twojej starej funkcji
saveGame = function() {
    // Najpierw niech zrobi stary zapis
    if (typeof originalSave === "function") originalSave();
    
    // Potem dopisujemy naszych nowych pracowników do pamięci telefonu
    const extraData = {
        hiredStaff: state.hiredStaff,
        selectedVet: state.clinic.staff
    };
    localStorage.setItem('vetExtraSave', JSON.stringify(extraData));
};

// 2. ROZSZERZAMY FUNKCJĘ WCZYTYWANIA
const originalLoad = loadGame; // kopia starego wczytywania
loadGame = function() {
    if (typeof originalLoad === "function") originalLoad();
    
    // Wczytujemy nasze dodatkowe dane
    const savedExtra = localStorage.getItem('vetExtraSave');
    if (savedExtra) {
        const parsed = JSON.parse(savedExtra);
        state.hiredStaff = parsed.hiredStaff || [];
        state.clinic.staff = parsed.selectedVet || "";
        
        // Odświeżamy widok, żeby pracownicy od razu byli widoczni
        if (typeof renderClinicApp === "function") renderClinicApp();
        if (typeof renderClinicBuilding === "function") renderClinicBuilding();
    }
};

// Zastąp tamten poprzedni kod na końcu tym:
window.addEventListener('beforeunload', () => {
    if (typeof saveGame === "function") saveGame();
});

let checkCount = 0;
const forceLoadInterval = setInterval(() => {
    const saved = localStorage.getItem('emergency_vet_save');
    if (saved) {
        const parsed = JSON.parse(saved);
        if (parsed.hired && parsed.hired.length > 0) {
            state.hiredStaff = parsed.hired;
            state.clinic.staff = parsed.selected;
            if (typeof renderClinicBuilding === "function") renderClinicBuilding();
        }
    }
    checkCount++;
    if (checkCount > 3) clearInterval(forceLoadInterval);
}, 1000);

function renderClinicBuilding() {
    // 1. Aktualizacja zapasów leków
    const medStock = (state.inventory && state.inventory.medical) ? state.inventory.medical : 0;
    const medStockEl = document.getElementById('clinic-bld-med-stock');
    if (medStockEl) medStockEl.innerText = `💊 Dostępne zestawy: ${medStock}`;
    
    const vList = document.getElementById('vets-list');
    if (!vList) return;
    vList.innerHTML = "";

    // 2. NOWOŚĆ: Szukamy WSZYSTKICH zaznaczonych pracowników
    // Sprawdzamy kogo zaznaczyliśmy kropkami w Laptopie
    if (!state.clinic.currentStaff || state.clinic.currentStaff.length === 0) {
        vList.innerHTML = `<p style="color:#666; font-size:11px; text-align:center; padding:20px;">Brak przypisanego personelu.<br>Wybierz zespół w Laptopie.</p>`;
        return;
    }

    // 3. Generujemy karty dla każdego zaznaczonego pracownika
    state.clinic.currentStaff.forEach(selectedName => {
        // Znajdujemy pełne dane pracownika w hiredStaff na podstawie imienia
        const v = state.hiredStaff.find(worker => worker.name === selectedName);
        
        // Jeśli pracownik istnieje w kadrze, rysujemy jego kartę
        if (v) {
            const now = Date.now();
            const isBusy = v.busyUntil > now;
            const syringes = "💉".repeat(v.exp || 1);
            const prices = { 1: 500, 2: 1000, 3: 1500, 4: 2000 };
            const trainCost = prices[v.exp] || 2500;

            let statusLabel = "";
            if(isBusy) {
                const rem = v.busyUntil - now;
                const h = Math.floor(rem / (1000 * 60 * 60));
                const m = Math.floor((rem % (1000 * 60 * 60)) / (1000 * 60));
                const s = Math.floor((rem % (1000 * 60)) / 1000);
                const timeStr = `${h > 0 ? h+'h ' : ''}${m}m ${s}s`;
                
                if(v.busyType === "urlop") statusLabel = `<span style="color:var(--orange); font-weight:bold; font-size:9px;">[Urlop - ${timeStr}]</span>`;
                else if(v.busyType === "szkolenie") statusLabel = `<span style="color:var(--gold); font-weight:bold; font-size:9px;">[Szkolenie - ${timeStr}]</span>`;
                else statusLabel = `<span style="color:var(--red); font-weight:bold; font-size:9px;">[W pracy - ${timeStr}]</span>`;
            }

            // Dodajemy kolejną kartę do listy (+=)
            vList.innerHTML += `
                <div class="vet-card" style="margin-bottom: 10px;">
                    <div style="display:flex; justify-content:space-between; align-items:center;">
                        <strong>${v.name}</strong>
                        <span class="syringes">${syringes}</span>
                    </div>
                    <div class="vet-stats">
                        <span>Zadowolenie:</span>
                        <div class="p-bar" style="width:50px; height:6px; margin:0;">
                            <div class="p-fill" style="width:${v.mood || 100}%; background:${(v.mood || 100) < 30 ? 'var(--red)' : 'var(--green)'}"></div>
                        </div>
                        ${statusLabel}
                    </div>
                    <div class="vet-actions">
                        <button class="btn-vet" style="background:var(--accent-blue)" onclick="vetAction(${v.id}, 'urlop')" ${isBusy ? 'disabled' : ''}>URLOP</button>
                        <button class="btn-vet" style="background:var(--gold); color:#000" onclick="vetAction(${v.id}, 'szkolenie')" ${isBusy || v.exp >= 5 ? 'disabled' : ''}>SZKOLENIE (${trainCost}zł)</button>
                        <button class="btn-vet" style="background:var(--red)" onclick="vetAction(${v.id}, 'fire')">ZWOLNIJ</button>
                    </div>
                </div>`;
        }
    });
}

function toggleMultiDropdown() {
    const menu = document.getElementById('clinic-staff-multi');
    if (!menu) return;

    if (menu.style.display === 'none' || menu.style.display === '') {
        menu.style.display = 'block';
    } else {
        menu.style.display = 'none';
    }
}

function toggleStaff(name) {
    // Jeśli nie ma jeszcze listy w pamięci, stwórz ją
    if (!state.clinic.currentStaff) {
        state.clinic.currentStaff = [];
    }

    const index = state.clinic.currentStaff.indexOf(name);
    
    if (index === -1) {
        state.clinic.currentStaff.push(name); // Dodaj pracownika
    } else {
        state.clinic.currentStaff.splice(index, 1); // Usuń pracownika
    }

    // To jest klucz do zapisu!
    localStorage.setItem('shelterGame', JSON.stringify(state));
    
    if (typeof saveGame === 'function') saveGame();

    renderClinicApp(); // Odśwież widok
}

function updateProfileStats() {
    // 1. Liczymy pracowników
    const staffCount = state.hiredStaff ? state.hiredStaff.length : 0;

    // 2. Liczymy koszty pracowników (dokładnie tak, jak w Twoim systemie wypłat)
    let totalSalaries = 0;
    if (state.hiredStaff && state.hiredStaff.length > 0) {
        totalSalaries = state.hiredStaff.reduce((sum, s) => sum + (s.dailyRate || 0), 0);
    }

    // 3. Liczymy psy
    let dogCount = 0;
    if (state.boxes) {
        state.boxes.forEach(box => {
            if (box.dogs) dogCount += box.dogs.length;
        });
    }

    // 4. Sumujemy wszystko
    const totalDaily = totalSalaries;

    // 5. Wpisujemy do HTML (do Twoich ID)
    const sEl = document.getElementById('display-staff');
    const dEl = document.getElementById('display-dogs');
    const cEl = document.getElementById('display-costs');

    if (sEl) sEl.innerText = staffCount + " os.";
    if (dEl) dEl.innerText = dogCount;
    if (cEl) cEl.innerText = totalDaily + " zł";
// 6. Aktualizujemy staż managera
    const daysEl = document.getElementById('display-days');
    if (daysEl) {
        daysEl.innerText = `Staż managera: ${state.daysPlayed || 0} dni`;
    }
}
